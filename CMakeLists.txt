cmake_minimum_required(VERSION 3.21)
project(HpMGARD-Tools LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

# The compression library (CompressHpMGARDOperator install prefix: -Dhpmgard_DIR=<prefix>/lib64/cmake/hpmgard)
find_package(hpmgard REQUIRED)
option(HPMT_BUILD_ADIOS2_TOOLS "Build the ADIOS2 drivers (needs ADIOS2)" OFF)
set(HPMGARD_PLUGIN_DIR "" CACHE PATH "Directory holding libCompressHpMGARDOperator.so (for the plugin test)")

add_executable(hpmgard_bench hpmgard_bench.cpp)
target_link_libraries(hpmgard_bench PRIVATE hpmgard::hpmgard)

enable_testing()
foreach(t test_quadrature test_operators test_codecs test_compressor)
  add_executable(${t} tests/${t}.cpp)
  target_link_libraries(${t} PRIVATE hpmgard::hpmgard)
  add_test(NAME ${t} COMMAND ${t})
endforeach()

if(HPMT_BUILD_ADIOS2_TOOLS)
  find_package(ADIOS2 REQUIRED)
  foreach(t hpmgard_adios_roundtrip hpmgard_adios_compress hpmgard_adios_decompress hpmgard_verify hpmgard_meta_dump)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/${t}.cpp")
      add_executable(${t} ${t}.cpp)
      target_link_libraries(${t} PRIVATE hpmgard::hpmgard adios2::cxx)
      if(t STREQUAL "hpmgard_adios_roundtrip")
        add_test(NAME adios_roundtrip COMMAND ${t} ${CMAKE_CURRENT_BINARY_DIR}/roundtrip_test.bp)
        set_tests_properties(adios_roundtrip PROPERTIES ENVIRONMENT "ADIOS2_PLUGIN_PATH=${HPMGARD_PLUGIN_DIR}")
      endif()
    endif()
  endforeach()
endif()
